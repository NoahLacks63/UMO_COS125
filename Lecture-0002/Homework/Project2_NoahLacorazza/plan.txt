def find_min(d):
    items = list(d.items())

    if len(items) == 1:
        return items[0][1]

    key, value = items[0]
    rest_min = find_min(dict(items[1:]))

    if value < rest_min:
        return value
    else:
        return rest_min

def find_max(d):

    if len(items) == 1:

def write_log(message, filename):
    try:
        with open(filename, "x") as f:
            f.write(message)

    except FileExistsError:
        write_log(message, input("Try a new file name: ))
